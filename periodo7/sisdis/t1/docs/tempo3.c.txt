#include <stdio.h>
#include <stdlib.h>
#include "smpl.h"

#define test     1
#define fault    2
#define recovery 3

typedef struct {
    int id;          /* facility do SMPL */
    int *State;      /* vetor State[N] local deste processo */
} TipoProcesso;

TipoProcesso *processo;
int N;

void imprime_state(int i) {
    int j;
    printf("        State[%d] = [", i);
    for (j = 0; j < N; j++) {
        if (processo[i].State[j] < 0) printf(" -1");
        else                          printf("  %d", processo[i].State[j]);
        if (j < N - 1) printf(",");
    }
    printf(" ]\n");
}

int main(int argc, char *argv[]) {
    static int N_, token, event, r, i, j;
    static int MaxTempoSimulac = 120;
    static char fa_name[5];

    if (argc != 2) {
        puts("uso correto: ./tempo3 <numero de processos>");
        exit(1);
    }
    N = atoi(argv[1]);

    smpl(0, "TP0 Tarefa 3");
    reset();
    stream(1);

    processo = (TipoProcesso *)malloc(sizeof(TipoProcesso) * N);
    for (i = 0; i < N; i++) {
        memset(fa_name, '\0', 5);
        sprintf(fa_name, "%d", i);
        processo[i].id    = facility(fa_name, 1);
        processo[i].State = (int *)malloc(sizeof(int) * N);
        for (j = 0; j < N; j++) processo[i].State[j] = -1;  /* unknown */
        processo[i].State[i] = 0;                            /* a si proprio: correto */
    }

    for (i = 0; i < N; i++) schedule(test, 30.0, i);
    schedule(fault, 31.0, 1);
    schedule(fault, 31.0, 3);
    schedule(recovery, 91.0, 1);

    while (time() < MaxTempoSimulac) {
        cause(&event, &token);
        switch (event) {
            case test:
                if (status(processo[token].id) != 0) break; /* processo falho nao testa */
                printf("t=%5.1f  processo %d inicia intervalo de testes\n", time(), token);
                /* testa o anel ate achar um correto, atualizando State a cada teste */
                for (i = 1; i <= N; i++) {
                    j = (token + i) % N;
                    if (j == token) break;                   /* deu a volta: todos falhos */
                    if (status(processo[j].id) != 0) {       /* j esta falho */
                        processo[token].State[j] = 1;
                        printf("t=%5.1f  processo %d testou %d -> FALHO\n", time(), token, j);
                    } else {                                 /* j esta correto */
                        processo[token].State[j] = 0;
                        printf("t=%5.1f  processo %d testou %d -> CORRETO\n", time(), token, j);
                        break;                               /* para ao achar um correto */
                    }
                }
                imprime_state(token);
                schedule(test, 30.0, token);
                break;
            case fault:
                r = request(processo[token].id, token, 0);
                printf("t=%5.1f  >>> processo %d FALHOU\n", time(), token);
                break;
            case recovery:
                release(processo[token].id, token);
                printf("t=%5.1f  >>> processo %d RECUPEROU\n", time(), token);
                schedule(test, 1.0, token);
                break;
        }
    }
    return 0;
}
