#include <stdio.h>
#include <stdlib.h>
#include "smpl.h"

#define test     1
#define fault    2
#define recovery 3

typedef struct {
    int id;          /* facility do SMPL */
    int *State;      /* vetor State[N] local deste processo */
} TipoProcesso;

TipoProcesso *processo;
int N;

void imprime_state(int i) {
    int j;
    printf("        State[%d] = [", i);
    for (j = 0; j < N; j++) {
        if (processo[i].State[j] < 0) printf(" -1");
        else                          printf("  %d", processo[i].State[j]);
        if (j < N - 1) printf(",");
    }
    printf(" ]\n");
}

int main(int argc, char *argv[]) {
    int  token, event, r, i, j, k;
    int  MaxTempoSimulac = 150;
    int  *testado;       /* marca quais processos foram testados NESTA rodada */
    char fa_name[5];

    if (argc != 2) {
        puts("uso correto: ./tempo4 <numero de processos>");
        exit(1);
    }
    N = atoi(argv[1]);

    smpl(0, "TP0 Tarefa 4");
    reset();
    stream(1);

    processo = (TipoProcesso *)malloc(sizeof(TipoProcesso) * N);
    testado  = (int *)malloc(sizeof(int) * N);
    for (i = 0; i < N; i++) {
        memset(fa_name, '\0', 5);
        sprintf(fa_name, "%d", i);
        processo[i].id    = facility(fa_name, 1);
        processo[i].State = (int *)malloc(sizeof(int) * N);
        for (j = 0; j < N; j++) processo[i].State[j] = -1;
        processo[i].State[i] = 0;
    }

    for (i = 0; i < N; i++) schedule(test, 30.0, i);
    schedule(fault, 31.0, 2);
    schedule(recovery, 121.0, 2);

    while (time() < MaxTempoSimulac) {
        cause(&event, &token);
        switch (event) {
            case test:
                if (status(processo[token].id) != 0) break;
                printf("t=%5.1f  processo %d inicia intervalo de testes\n", time(), token);

                for (k = 0; k < N; k++) testado[k] = 0;   /* zera marcacao da rodada */

                /* testa o anel ate achar um correto */
                for (i = 1; i <= N; i++) {
                    j = (token + i) % N;
                    if (j == token) break;                /* todos falhos */
                    testado[j] = 1;
                    if (status(processo[j].id) != 0) {    /* j falho */
                        processo[token].State[j] = 1;
                        printf("t=%5.1f  processo %d testou %d -> FALHO\n", time(), token, j);
                    } else {                              /* j correto: testa e DISSEMINA */
                        processo[token].State[j] = 0;
                        printf("t=%5.1f  processo %d testou %d -> CORRETO"
                               " (obtem dele o estado dos demais)\n", time(), token, j);
                        /* copia de j o estado dos processos que NAO testou nesta
                         * rodada e que nao sejam o proprio testador */
                        for (k = 0; k < N; k++) {
                            if (k == token) continue;     /* nao sobre si mesmo */
                            if (testado[k])  continue;     /* nao sobre os testados nesta rodada */
                            if (processo[j].State[k] != -1)
                                processo[token].State[k] = processo[j].State[k];
                        }
                        break;                            /* para ao achar um correto */
                    }
                }
                imprime_state(token);
                schedule(test, 30.0, token);
                break;
            case fault:
                r = request(processo[token].id, token, 0);
                printf("t=%5.1f  >>> processo %d FALHOU\n", time(), token);
                break;
            case recovery:
                release(processo[token].id, token);
                printf("t=%5.1f  >>> processo %d RECUPEROU\n", time(), token);
                schedule(test, 1.0, token);
                break;
        }
    }
    return 0;
}
